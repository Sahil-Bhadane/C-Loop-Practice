#include <stdio.h>
int main()
{
    int a,isprime = 1,i = 1;
    printf("Enter a number:");
    scanf("%d", & a );
    for(i=2;i*i<=a;i++){
        if( a % i == 0){
            isprime=0;
            break;
        }
    }
    if(isprime==1){
        printf("%d is a prime number", a );
    }
    else{
        printf("%d is not a prime number", a );
    }
    return 0;
}
